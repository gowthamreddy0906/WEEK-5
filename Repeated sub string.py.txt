REPEATED SUB-STRING

class Solution {
    public boolean repeatedSubstringPattern(String s) {
        int n = s.length();
        
        // The largest possible repeating substring is half the length of s
        for (int i = n / 2; i >= 1; i--) {
            // Check if the current length divides the total length perfectly
            if (n % i == 0) {
                int repeats = n / i;
                String sub = s.substring(0, i);
                StringBuilder sb = new StringBuilder();
                
                for (int j = 0; j < repeats; j++) {
                    sb.append(sub);
                }
                
                if (sb.toString().equals(s)) {
                    return true;
                }
            }
        }
        return false;
    }
}
